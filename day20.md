		# 查询出版社

		# 获取post发送过来的数据,转化成字典

		print(dic)
		# 添加数据到 book表

		data = Book.objects.create(**dic)

		# data.authors.add(1,2)
		data.authors.add(*lst) # *号的解包操作
		# 查询刚才关系表添加的数据条数
		res = data.authors.filter().count()
		print(res)
		# data.id代表的是添加的数据id , res代表的是添加的数据条数;都为真跳转首页

			return redirect("/book/index/")

			复习:* 和 ** 的解包用法;

				def func(*args,**kwargs):

def delete(request,del_id):
	# 删除之前,先跳一个页面确认再删
	if request.method == "GET":
		data = Book.objects.get(pk=del_id)
		return render(request,"delete.html",{"data":data})

	elif request.method == "POST":
		# 查询单条数据
		data = Book.objects.filter(id=del_id)
		# 删除 , 返回删除的条数
		res = data.delete()
		print(res)
		if res:
			return redirect("/book/index/")
		else:
			return HttpResponse("删除失败了")
